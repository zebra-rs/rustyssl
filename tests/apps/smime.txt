# `openssl smime` (apps/smime.c): run with upstream's tool on the reference, then ours on
# rustyssl and on the reference (cargo xtask test-apps); output and exit codes must match.
# Signatures without attributes (-noattr) by RSA keys are deterministic and byte-compared;
# signing times, DSA and ECDSA signatures, S/MIME boundaries and encryption are random, so
# those outputs go to files that the following lines verify or decrypt.
#
# Deterministic signatures.
smime -sign -in {src}/test/smcont.txt -outform PEM -nodetach -noattr -signer {src}/test/smime-certs/smrsa1.pem

smime -verify -in {dir}/s1.der -inform DER -CAfile {src}/test/smime-certs/smroot.pem

smime -sign -in {src}/test/smcont.txt -outform PEM -nodetach -noattr -signer {src}/test/smime-certs/smrsa1.pem -certfile {src}/test/smime-certs/smroot.pem -nocerts
